# 1. Go into your local project folder
cd /path/to/your/local/folder

# 2. Initialize a new Git repository (only do this once)
git init

# 3. Add all files and folders (or add specific files if you prefer)
git add .

# 4. Make your first commit
git commit -m "Initial commit"

# 5. Rename the default branch to 'main' (modern standard)
git branch -M main

# 6. Create a new empty repository on GitHub / GitLab etc.
#    → Do NOT initialize it with README, .gitignore, or license
#    → Copy the repository URL (it looks like https://github.com/yourname/your-repo.git)

# 7. Connect your local repo to the remote one
git remote add origin https://github.com/yourname/your-repo.git

# 8. Push your files to the remote repository
git push -u origin main
